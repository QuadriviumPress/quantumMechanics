%% Print edition of *Quantum Mechanics*.
%%
%% A local jtex template, referenced from myst.yml as `template: ./templates/book`.
%% It began as MyST's stock `plain_latex_book` and diverges from it in five ways,
%% each of which is load-bearing:
%%
%%   1. The problem-set commands that plugins/export.mjs brackets every exercise
%%      and solution with. Neither may be boxed -- see the comment there.
%%   2. Sectioning depth pinned to match the website's numbering exactly.
%%   3. Page geometry and running heads for something meant to be printed.
%%   4. A copyright page in \frontmatter, because index.md is not part of the
%%      printed book.
%%   5. The `chapter` option, which turns the same source into a standalone
%%      single-chapter offprint numbered as it is in the full book.
%%
%% Built with XeLaTeX (MyST's default: `latexmk -f -xelatex`), which is what lets
%% raw Unicode in the source -- ℏ, ψ, Δ, °, ×, ⊕ -- pass through without a
%% preamble fight.

[# if options.chapter #]
%% One chapter, standalone. `article` rather than `book`: a one-article export
%% emits no title heading, so the chapter's `##` reading sections arrive as
%% `\section` and there is no `\chapter` left to host. The chapter number is
%% restored to the counters further down.
\documentclass[letterpaper,11pt]{article}
[# else #]
%% oneside/openany, and the margins below, are for a student printing the book
%% single-sided on a classroom printer: no blank verso pages, no gutter offset,
%% and a measure of about 6.25in -- roughly 75 characters, which is what a
%% textbook wants.
\documentclass[letterpaper,11pt,oneside,openany]{book}
[# endif #]

\usepackage[letterpaper,top=1in,bottom=1in,left=1.15in,right=1.15in,
            headheight=14pt]{geometry}

%% XeLaTeX, so fontspec rather than the pdfTeX pair fontenc+inputenc: under T1
%% the 8-bit EC fonts have no slot for the Unicode the source is written in, and
%% the Greek letters, en dashes, and ℏ of quantum mechanics all come out as
%% silent "Missing character" warnings and blank space on the page.
\usepackage{fontspec}
%% Latin Modern ships its small caps as a separate face; without naming it,
%% \textsc silently falls back to upright and the title page loses its styling.
\setmainfont{Latin Modern Roman}[SmallCapsFont={Latin Modern Roman Caps}]
\setsansfont{Latin Modern Sans}
\setmonofont{Latin Modern Mono}

%% U+2113 SCRIPT SMALL L and U+210F PLANCK CONSTANT OVER TWO PI appear
%% throughout this book, and no text font carries either; route them to the
%% math symbols they mean.
\usepackage{newunicodechar}
\newunicodechar{ℓ}{\ensuremath{\ell}}
\newunicodechar{ℏ}{\ensuremath{\hbar}}

\usepackage{microtype}
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{graphicx}
\usepackage{caption}
%% Captions in smaller type: a full attribution sentence at body size can push
%% a float past the height of the page.
\captionsetup{font=small,labelfont=bf,skip=6pt}
\usepackage{booktabs}
\usepackage{natbib}
\usepackage{xcolor}
\usepackage{changepage}
\usepackage{framed}
\usepackage{float}
\usepackage{needspace}
\usepackage{fancyhdr}
\usepackage{hyperref}
\bibliographystyle{abbrvnat}

% Make list items more compact
\usepackage{enumitem}
\setlist[itemize]{noitemsep, topsep=0pt}

[- IMPORTS -]

%% ---------------------------------------------------------------------------
%% Numbering
%%
%% The website numbers chapters and `##` reading sections (4, 4.1, 4.2 ...) and
%% leaves everything deeper unnumbered; `outline_maxdepth: 2` in myst.yml keeps
%% its outline to the same two levels. These counters are the LaTeX spelling of
%% exactly that, so "Section 4.2" means the same thing in every edition. Without
%% them LaTeX would number the `###` headings too, and "Learning Objectives"
%% would come out as 4.0.1.
%% ---------------------------------------------------------------------------
\setcounter{secnumdepth}{1}
\setcounter{tocdepth}{1}
[# if options.chapter #]
% A one-article export carries no title heading, so the `##` reading sections
% arrive as `\section` and everything below shifts up with them. Prefixing the
% chapter number by hand is what makes an offprint's 4.1, Figure 4.3 and
% equation (4.47) the same 4.1, Figure 4.3 and (4.47) as the full book's --
% which matters, because the prose refers to them by number.
\renewcommand{\thesection}{[-options.chapter-].\arabic{section}}
\renewcommand{\thesubsection}{\thesection.\arabic{subsection}}
\renewcommand{\thefigure}{[-options.chapter-].\arabic{figure}}
\renewcommand{\thetable}{[-options.chapter-].\arabic{table}}
\renewcommand{\theequation}{[-options.chapter-].\arabic{equation}}
[# endif #]

%% ---------------------------------------------------------------------------
%% Problem sets
%%
%% plugins/export.mjs renders every {exercise} and {solution} as a bold run-in
%% title followed by the body, bracketed by the commands below. Two constraints
%% shaped that:
%%
%%   - No box. About a quarter of the worked solutions contain a {figure}, and
%%     LaTeX refuses to open a float inside framed, minipage, or any other box:
%%     it fails with "Not in outer par mode" and the figure, its caption, and
%%     every {numref} pointing at it are lost.
%%   - No custom environment. The same .tex feeds pandoc for the DOCX edition,
%%     and pandoc discards the entire contents of any environment it does not
%%     know. Bare commands it cannot read are skipped harmlessly instead, so the
%%     text between them survives into Word.
%%
%% \leftskip rather than a list: floats escape a group unharmed.
%% ---------------------------------------------------------------------------
\newcommand{\mystexercisestart}{\par\needspace{3\baselineskip}\medskip\noindent}
\newcommand{\mystexerciseend}{\par\medskip}
\newcommand{\mystsolutionstart}{\par\needspace{3\baselineskip}\smallskip
  \begingroup\leftskip=1.2em\relax\noindent}
\newcommand{\mystsolutionend}{\par\endgroup\medskip}

%% Quotes -- and, by way of plugins/export.mjs, {margin} notes. The web theme
%% floats a margin note beside the text; a printed page this wide has no margin
%% to float into, so it becomes a rule-and-tint aside in the flow. pandoc reads
%% \begin{quote} natively, so it survives into Word as a block quotation.
\renewenvironment{quote}{%
  \def\FrameCommand{%
    \hspace{1pt}%
    {\color{darkblue}\vrule width 2pt}%
    {\color{quoteshade}\vrule width 4pt}%
    \colorbox{quoteshade}%
  }%
  \MakeFramed {\advance\hsize-\width \FrameRestore}%
  \noindent\hspace{-8pt}% disable indenting first paragraph
  \begin{adjustwidth}{0pt}{0pt}% adjust as needed
  \vspace{2pt}\vspace{2pt}%
}
{%
  \vspace{2pt}\end{adjustwidth}\endMakeFramed%
}
